<script>
  let currentCat = 'all';

  function updateCounter() {
    const visible = document.querySelectorAll('.word-item:not(.hidden)').length;
    const total = document.querySelectorAll('.word-item').length;
    document.getElementById('counter').textContent = visible + ' ta so\'z ko\'rsatilmoqda (jami: ' + total + ')';
  }

  function filterWords() {
    const query = document.getElementById('searchInput').value.toLowerCase().trim();
    const items = document.querySelectorAll('.word-item');
    const sections = document.querySelectorAll('.section-title');
    let anyVisible = false;

    items.forEach(item => {
      const cat = item.getAttribute('data-cat');
      const en = item.querySelector('.word-en').textContent.toLowerCase();
      const uz = item.querySelector('.word-uz').textContent.toLowerCase();
      const catMatch = currentCat === 'all' || cat === currentCat;
      const textMatch = query === '' || en.includes(query) || uz.includes(query);
      if (catMatch && textMatch) {
        item.classList.remove('hidden');
        anyVisible = true;
      } else {
        item.classList.add('hidden');
      }
    });

    sections.forEach(sec => {
      const cat = sec.getAttribute('data-cat');
      const nextList = sec.nextElementSibling;
      const hasVisible = nextList && nextList.querySelector('.word-item:not(.hidden)');
      sec.style.display = (currentCat === 'all' || cat === currentCat) && hasVisible ? '' : 'none';
      if (nextList) nextList.style.display = hasVisible ? '' : 'none';
    });

    document.getElementById('noResult').style.display = anyVisible ? 'none' : 'block';
    updateCounter();
  }

  function filterCategory(cat, btn) {
    currentCat = cat;
    document.querySelectorAll('.filter-bar button').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    document.getElementById('searchInput').value = '';
    filterWords();
  }

  updateCounter();
</script>
